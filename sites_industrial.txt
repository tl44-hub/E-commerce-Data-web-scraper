name: Run industrial scraper

on:
  workflow_dispatch:        # lets you press a Start button
  schedule:
    - cron: "0 6 * * 2"     # also runs by itself every Tuesday (the pet project runs Mondays)

permissions:
  contents: write           # lets the robot save files into your repository

concurrency:
  group: scrape-industrial  # never run two of these at once

env:
  DATA_DIR: data/industrial # all industrial results go to their own folder

jobs:
  scrape:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip

      - name: Install tools
        run: pip install -r requirements.txt

      - name: Run the scraper
        run: python main.py --no-manual --sites-file sites_industrial.txt --db industrial.db --max-attempts 3
        env:
          SCRAPER_API_KEY: ${{ secrets.SCRAPER_API_KEY }}
          # render=true makes Scrape.do load the page in a real browser (5 credits per page).
          # If the log shows blocks, change it to &super=true&render=true (25 credits per page).
          PROXY_URL_TEMPLATE: "https://api.scrape.do/?token={api_key}&url={url}&render=true"

      - name: Clean the offers
        if: always()
        continue-on-error: true
        run: python parts_export.py industrial.db

      - name: Match parts across distributors
        if: always()
        continue-on-error: true
        run: python parts_match.py

      - name: Save results to the repository
        if: always()
        run: |
          mkdir -p data/industrial
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data/industrial
          git diff --staged --quiet || git commit -m "Update industrial data $(date -u +%F)"
          git pull --rebase
          git push

      - name: Keep a copy as a download too
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: industrial-results
          path: |
            data/industrial/*.csv
            industrial.db*
